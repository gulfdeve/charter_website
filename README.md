import { Montserrat } from "next/font/google";
import "../globals.css";
import Navbar from "@/app/components/Navbar";
import FareHarborLoader from '@/app/components/FareHarborLoader';
import { LanguageProvider } from "../context/LanguageContext";
import { DataProvider } from "../context/DataContext";
import Sociallink from "@/app/components/Sociallink";
import FirstFooter from "@/app/components/FirstFooter";
import SecondFooter from "@/app/components/SecondFooter";
import CopyrightFooter from "@/app/components/CopyrightFooter";
import { NextIntlClientProvider } from "next-intl";
import { getLocale, getMessages } from "next-intl/server";
import { routing } from "../i18n/routing";
import { notFound } from "next/navigation";
import {FareHarborProvider} from "@/app/context/FareHarborContext"
import 'react-phone-input-2/lib/style.css';
import React from "react";


async function fetchData() {
  const response = await fetch("https://www.centauruscharter.com/api/api_new/basic");
  if (!response.ok) {
    throw new Error("Failed to fetch data");
  }
  return response.json();
}


const montserrat = Montserrat({
  subsets: ["latin"],
  variable: "--font-montserrat",
});

export const metadata = {
  title: "Create Next App",
  description: "Generated by create next app",
};

export default async function RootLayout({
  children,
  params,
 
}: {
  children: React.ReactNode;
 
  params: Promise<{ locale: string }>;
}) {
  let fetchedData = null;
  let isError = false;
  let isLoading = true;

  try {
    isLoading = true;
    fetchedData = await fetchData();
  } catch (error) {
    console.error("Error fetching data:", error);
    isError = true;
  } finally {
    isLoading = false;
  }
  const { locale } = await params;


  // Ensure that the incoming `locale` is valid
  if (!routing.locales.includes(locale as any)) {
    notFound();
  }

  const messages = await getMessages();

  return (
    <html lang="en">
      <body className={`${montserrat.variable}`}>
      {isLoading ? (
          <div>Loading...</div> // Show loading indicator while data is being fetched
        ) : isError ? (
          <div>Error fetching data</div> // Show error message if fetching fails
        ) : (
        <NextIntlClientProvider locale={locale} messages={messages}>
          <DataProvider>
            <LanguageProvider>
              <Navbar />
              <FareHarborProvider>
              {React.cloneElement(children as React.ReactElement, { fetchedData })}
              </FareHarborProvider>
              <Sociallink />
              <FirstFooter />
              <SecondFooter />
              <CopyrightFooter />
              <FareHarborLoader />
            </LanguageProvider>
          </DataProvider>
        </NextIntlClientProvider>
        )}
      </body>
    </html>
  );
}
